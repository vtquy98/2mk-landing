(function (root) {
  'use strict';
  if (root.TM && root.TM.__v) return; // idempotent : double inclusion sans effet
  var TM = {};
  TM.__v = '1.0.0';
  /* ---------------------------------------------------------------------
   * 1. NOMBRES ET FORMATAGE (locale fr-FR)
   * ------------------------------------------------------------------- */
  /**
   * Parse une saisie utilisateur française en Number.
   * Accepte : "1 234,56", "1234.56", "1.234,56", "12 €", "3,5 %", "" -> null
   * Retourne null si la valeur n'est pas un nombre exploitable — jamais NaN,
   * pour que l'appelant distingue « vide » de « zéro ».
   */
  TM.parseNum = function (value) {
    if (value === null || value === undefined) return null;
    if (typeof value === 'number') return isFinite(value) ? value : null;
    var s = String(value).trim();
    if (!s) return null;
    // Supprime espaces (dont insécables), symboles monétaires et %
    s = s.replace(/[\s  ]/g, '').replace(/[€$%]/g, '');
    var hasComma = s.indexOf(',') !== -1;
    var hasDot = s.indexOf('.') !== -1;
    if (hasComma && hasDot) {
      // Le dernier séparateur rencontré est le séparateur décimal
      s = s.lastIndexOf(',') > s.lastIndexOf('.')
        ? s.replace(/\./g, '').replace(',', '.')   // 1.234,56
        : s.replace(/,/g, '');                     // 1,234.56
    } else if (hasComma) {
      s = s.replace(',', '.');
    }
    var n = Number(s);
    return isFinite(n) ? n : null;
  };
  var _fmt = function (opts) {
    try { return new Intl.NumberFormat('fr-FR', opts); } catch (e) { return null; }
  };
  var _eur0 = _fmt({ style: 'currency', currency: 'EUR', minimumFractionDigits: 0, maximumFractionDigits: 0 });
  var _eur2 = _fmt({ style: 'currency', currency: 'EUR', minimumFractionDigits: 2, maximumFractionDigits: 2 });
  /** Montant en euros. decimals = 0 (défaut) ou 2. */
  TM.eur = function (n, decimals) {
    if (n === null || n === undefined || !isFinite(n)) return '—';
    var f = decimals === 2 ? _eur2 : _eur0;
    if (f) return f.format(n);
    return TM.num(n, decimals === 2 ? 2 : 0) + ' €';
  };
  /** Nombre brut, d décimales, séparateur de milliers fr. */
  TM.num = function (n, d) {
    if (n === null || n === undefined || !isFinite(n)) return '—';
    d = d === undefined ? 2 : d;
    var f = _fmt({ minimumFractionDigits: d, maximumFractionDigits: d });
    if (f) return f.format(n);
    return n.toFixed(d).replace('.', ',');
  };

   * Pourcentage. `rate` est une FRACTION décimale (0.0512 -> "5,12 %").
   * Convention interne : tous les taux circulent en fraction, jamais en points.
   * On ne convertit qu'à l'affichage — c'est la source n°1 d'erreurs d'un facteur 100.
   */
  TM.pct = function (rate, d) {
    if (rate === null || rate === undefined || !isFinite(rate)) return '—';
    d = d === undefined ? 2 : d;
    return TM.num(rate * 100, d) + ' %';
  };
  /** Arrondi commercial à d décimales, robuste au flottant (0.145 -> 0.15). */
  TM.round = function (n, d) {
    if (!isFinite(n)) return n;
    d = d === undefined ? 2 : d;
    var p = Math.pow(10, d);
    return Math.round((n * p + (n >= 0 ? 1 : -1) * 1e-9)) / p;
  };
  TM.clamp = function (n, min, max) { return Math.min(Math.max(n, min), max); };
  /* ---------------------------------------------------------------------
   * 2. DATES
   * ------------------------------------------------------------------- */
  /** "2026-03-31" ou Date -> Date UTC à minuit (évite les dérives de fuseau). */
  TM.date = function (v) {
    if (v instanceof Date) return new Date(Date.UTC(v.getFullYear(), v.getMonth(), v.getDate()));
    var m = /^(\d{4})-(\d{2})-(\d{2})/.exec(String(v));
    if (m) return new Date(Date.UTC(+m[1], +m[2] - 1, +m[3]));
    var m2 = /^(\d{2})\/(\d{2})\/(\d{4})$/.exec(String(v).trim()); // jj/mm/aaaa
    if (m2) return new Date(Date.UTC(+m2[3], +m2[2] - 1, +m2[1]));
    var d = new Date(v);
    return isNaN(d.getTime()) ? null : new Date(Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()));
  };
  /** Nombre de jours calendaires entre deux dates (Actual). */
  TM.days = function (from, to) {
    var a = TM.date(from), b = TM.date(to);
    if (!a || !b) return null;
    return Math.round((b - a) / 86400000);
  };
  /** Ajoute n mois en conservant la fin de mois (31 janv + 1 mois = 28/29 févr). */
  TM.addMonths = function (d, n) {
    var x = TM.date(d);
    var day = x.getUTCDate();
    var t = new Date(Date.UTC(x.getUTCFullYear(), x.getUTCMonth() + n, 1));
    var last = new Date(Date.UTC(t.getUTCFullYear(), t.getUTCMonth() + 1, 0)).getUTCDate();
    t.setUTCDate(Math.min(day, last));
    return t;
  };
  /* ---------------------------------------------------------------------
   * 3. FINANCE
   * Convention : un flux = { date: Date|"YYYY-MM-DD", amount: Number }
   * amount < 0 = décaissement investisseur, amount > 0 = encaissement.
   * ------------------------------------------------------------------- */
  /** Valeur actuelle nette d'une série de flux datés, au taux annuel `rate`.
   *  Actualisation Actual/365 — même convention qu'Excel XIRR. */

    var t0 = TM.date(flows[0].date);

   * TRI de flux datés (équivalent Excel XIRR), en fraction décimale.
   *
   * Méthode : Newton-Raphson amorcé à `guess`, avec repli sur une bisection
   * bornée si Newton diverge ou sort du domaine. La bisection garantit la
   * convergence dès qu'un changement de signe existe sur [-0,9999 ; 10] ;
   * c'est ce filet qui évite les « NaN » silencieux sur des flux atypiques
   * (revente à perte, dividendes irréguliers, durée < 1 an).
   *
   * Retourne null — jamais NaN — si le TRI n'est pas défini :
   *  - moins de 2 flux,
   *  - tous les flux de même signe (aucune racine),
   *  - aucun changement de signe de la VAN sur le domaine exploré.
   * Un null doit être affiché « non calculable », jamais 0 %.
   */
  TM.xirr = function (flows, guess) {
    if (!flows || flows.length < 2) return null;
    var f = flows.slice().sort(function (a, b) { return TM.date(a.date) - TM.date(b.date); });
    var hasNeg = false, hasPos = false;
    for (var i = 0; i < f.length; i++) {
      if (!isFinite(f[i].amount)) return null;
      if (f[i].amount < 0) hasNeg = true;
      if (f[i].amount > 0) hasPos = true;
    }
    if (!hasNeg || !hasPos) return null;
    var LO = -0.9999, HI = 10, TOL = 1e-9, r = (guess === undefined ? 0.1 : guess);
    // --- Newton-Raphson
    for (var k = 0; k < 60; k++) {
      var t0 = TM.date(f[0].date), npv = 0, d1 = 0;
      for (var j = 0; j < f.length; j++) {
        var t = TM.days(t0, f[j].date) / 365;
        var disc = Math.pow(1 + r, t);
        npv += f[j].amount / disc;
        d1 -= t * f[j].amount / (disc * (1 + r));
      }
      if (Math.abs(npv) < 1e-8) return r;
      if (!isFinite(d1) || d1 === 0) break;
      var next = r - npv / d1;
      if (!isFinite(next) || next <= LO || next > HI) break;
      if (Math.abs(next - r) < TOL) return next;
      r = next;
    }
    // --- Repli : bisection sur [LO, HI]
    var lo = LO, hi = HI;
    var flo = TM.xnpv(lo, f), fhi = TM.xnpv(hi, f);
    if (!isFinite(flo) || !isFinite(fhi) || flo * fhi > 0) return null;
    for (var b = 0; b < 300; b++) {
      var mid = (lo + hi) / 2, fm = TM.xnpv(mid, f);
      if (Math.abs(fm) < 1e-9 || (hi - lo) < 1e-10) return mid;
      if (flo * fm < 0) { hi = mid; fhi = fm; } else { lo = mid; flo = fm; }
    }
    return (lo + hi) / 2;
  };

    for (var i = 0; i < values.length; i++) {

      if (values[i] > 0) hasPos = true;

background:#e5e7eb; margin:18px 0; } </style><div class="s2m-container"><header><h2>Plus-value SCPI : guide complet 2026</h2><p>Comprendre, calculer et anticiper la fiscalité de vos plus-values sur parts de SCPI : barèmes, abattements par durée de détention, surtaxe, exemples chiffrés et points de vigilance. Mis à jour après l'adoption du PLF 2026 et de la LFSS 2026.</p><div class="s2m-badges"><span class="s2m-badge">Pédagogique</span> <span class="s2m-badge">Sans conseil en investissement</span> <span class="s2m-badge">À jour février 2026</span></div></header><p><a href="/guides-scpi-viager-demembrement">Guides</a> › Plus-value SCPI</p><h2>Comprendre la plus-value sur les SCPI</h2><p>La <strong>plus-value immobilière</strong> sur parts de SCPI correspond à la différence entre le prix de cession et le prix d'acquisition, frais d'acquisition inclus. Elle relève du <strong>régime des plus-values immobilières des particuliers</strong> (articles 150 U à 150 VH du CGI).</p><div class="s2m-note"><p>Les SCPI sont fiscalement transparentes : la plus-value sur les parts suit le même régime que la vente directe d'un bien immobilier. Le notaire ou la société de gestion calcule et prélève l'impôt à la source lors de la cession.</p></div><h3>Calcul de la plus-value brute</h3><p><strong>Plus-value brute = Prix de cession − Prix d'acquisition − Frais d'acquisition</strong></p><p>Le <strong>prix d'acquisition</strong> peut être majoré de :</p><ul><li><strong>Frais d'acquisition réels</strong> (droits d'enregistrement, frais de notaire à l'achat) ou, à défaut, un <strong>forfait de 7,5 %</strong> du prix d'achat</li><li><strong>Dépenses de travaux</strong> réellement supportées et justifiées, ou un <strong>forfait de 15 %</strong> du prix d'achat si le bien est détenu depuis <strong>plus de 5 ans</strong></li></ul><div class="s2m-note s2m-warning"><p><strong>Attention :</strong> Pour les parts de SCPI, le forfait travaux de 15 % n'est <strong>pas applicable</strong> car les associés ne supportent pas directement les dépenses de travaux. Seuls les frais d'acquisition réels (ou le forfait de 7,5 %) peuvent être déduits. Les travaux réalisés par la SCPI sont pris en compte dans la valeur comptable des parts.</p></div><h2>Fiscalité de la plus-value SCPI en 2026</h2><p>La plus-value nette (après abattements pour durée de détention) est soumise à <strong>trois prélèvements potentiels</strong>:</p><h3>1. Impôt sur le revenu (IR) – 19 %</h3><p>L'IR sur la plus-value immobilière est un prélèvement proportionnel au taux de <strong>19 %</strong>, après abattement pour durée de détention.</p><table class="s2m-table"><thead><tr><th>Durée de détention</th><th>Abattement IR</th><th>Abattement cumulé</th></tr></thead><tbody><tr><td>Moins de 6 ans</td><td>0 %</td><td>0 %</td></tr><tr><td>De la 6ᵉ à la 21ᵉ année</td><td>6 % par an</td><td>6 % à 96 %</td></tr><tr><td>22ᵉ année révolue</td><td>4 %</td><td><strong>100 %</strong></td></tr><tr><td>Au-delà de 22 ans</td><td>—</td><td><strong>Exonération totale IR</strong></td></tr></tbody></table><h3>2. Prélèvements sociaux (PS) – 17,2 %</h3><p>Les prélèvements sociaux s'appliquent au taux global de <strong>17,2 %</strong> (CSG 9,2 % + CRDS 0,5 % + prélèvement de solidarité 7,5 %), après un abattement distinct et plus long que pour l'IR.</p><table class="s2m-table"><thead><tr><th>Durée de détention</th><th>Abattement PS</th><th>Abattement cumulé</th></tr></thead><tbody><tr><td>Moins de 6 ans</td><td>0 %</td><td>0 %</td></tr><tr><td>De la 6ᵉ à la 21ᵉ année</td><td>1,65 % par an</td><td>1,65 % à 26,40 %</td></tr><tr><td>22ᵉ année révolue</td><td>1,60 %</td><td>28 %</td></tr><tr><td>De la 23ᵉ à la 30ᵉ année</td><td>9 % par an</td><td>37 % à 100 %</td></tr><tr><td>Au-delà de 30 ans</td><td>—</td><td><strong>Exonération totale PS</strong></td></tr></tbody></table><div class="s2m-note"><p><strong>LFSS 2026 :</strong> La loi de financement de la Sécurité sociale pour 2026 (loi n° 2025-1403 du 30 décembre 2025) a relevé les prélèvements sociaux à <strong>18,6 %</strong> pour certains revenus du capital (dividendes, plus-values mobilières, LMNP). Cependant, les <strong>plus-values immobilières</strong> et les <strong>revenus fonciers</strong> sont explicitement <strong>exclus</strong> de cette hausse et restent imposés au taux de <strong>17,2 %</strong>.</p></div><h3>3. Surtaxe sur les plus-values élevées</h3><p>Lorsque la plus-value nette imposable à l'IR (après abattement pour durée de détention) dépasse <strong>50 000 €</strong>, une <strong>taxe supplémentaire</strong> s'applique selon le barème progressif suivant :</p><table class="s2m-table"><thead><tr><th>Plus-value nette imposable (IR)</th><th>Taux de la surtaxe</th></tr></thead><tbody><tr><td>De 50 001 € à 60 000 €</td><td>2 %</td></tr><tr><td>De 60 001 € à 100 000 €</td><td>2 %</td></tr><tr><td>De 100 001 € à 110 000 €</td><td>3 %</td></tr><tr><td>De 110 001 € à 150 000 €</td><td>3 %</td></tr><tr><td>De 150 001 € à 160 000 €</td><td>4 %</td></tr><tr><td>De 160 001 € à 200 000 €</td><td>4 %</td></tr><tr><td>De 200 001 € à 210 000 €</td><td>5 %</td></tr><tr><td>De 210 001 € à 250 000 €</td><td>5 %</td></tr><tr><td>De 250 001 € à 260 000 €</td><td>6 %</td></tr><tr><td>Au-delà de 260 000 €</td><td>6 %</td></tr></tbody></table><p>Un mécanisme de lissage s'applique dans chaque tranche pour éviter les effets de seuil. Cette surtaxe disparaît lorsque la plus-value est totalement exonérée d'IR (après 22 ans).</p><div class="s2m-note s2m-warning"><p><strong>Taux maximum combiné :</strong> Pour une cession avant 6 ans de détention avec une plus-value nette imposable supérieure à 260 000 €, le taux global peut atteindre <strong>19 % + 17,2 % + 6 % = 42,2 %</strong>.</p></div><h2>Exemple de calcul complet</h2><h3>Situation</h3><ul><li>Prix d'achat : 10 000 € (il y a 8 ans)</li><li>Frais d'acquisition : forfait 7,5 % = 750 €</li><li>Prix d'acquisition majoré : 10 750 €</li><li>Prix de vente : 12 000 €</li><li><strong>Plus-value brute : 12 000 − 10 750 = 1 250 €</strong></li></ul><h3>Abattements après 8 ans de détention</h3><ul><li>IR : 3 années au-delà de la 5ᵉ année × 6 % = <strong>18 % d'abattement</strong></li><li>PS : 3 années au-delà de la 5ᵉ année × 1,65 % = <strong>4,95 % d'abattement</strong></li></ul><h3>Calcul de l'imposition</h3><div class="s2m-note"><p><strong>Impôt sur le revenu :</strong></p><ul><li>PV imposable IR : 1 250 € × (1 − 0,18) = 1 025 €</li><li>IR à payer : 1 025 € × 19 % = <strong>194,75 €</strong></li></ul><p><strong>Prélèvements sociaux :</strong></p><ul><li>PV imposable PS : 1 250 € × (1 − 0,0495) = 1 188,13 €</li><li>PS à payer : 1 188,13 € × 17,2 % = <strong>204,36 €</strong></li></ul><p><strong>Surtaxe :</strong> non applicable (PV nette IR &lt; 50 000 €)</p></div><p><strong>Total impôts : 399,11 €</strong> (IR + PS)</p><p><strong>Plus-value nette après impôts : 850,89 €</strong></p><div class="s2m-note s2m-success"><p><strong>Impact du forfait 7,5 % :</strong> Sans la majoration forfaitaire des frais d'acquisition, la plus-value brute serait de 2 000 € au lieu de 1 250 €, soit un impôt total de 638,57 € au lieu de 399,11 €. <strong>Économie : 239,46 €</strong>.</p></div><div class="s2m-note s2m-success"><p><strong>Effet de la durée de détention :</strong></p><ul><li>À 22 ans → IR = 0 € (seuls les PS restent dus)</li><li>À 30 ans → IR + PS = 0 € (exonération totale)</li></ul></div><h2>Performance globale : plus-value + revenus</h2><p>Pour apprécier la performance totale de votre investissement en SCPI, il faut combiner la <strong>plus-value sur les parts</strong> et les <strong>revenus distribués</strong> sur toute la période de détention.</p><h3>Exemple sur 8 ans (avec forfait 7,5 %)</h3><ul><li>Capital initial investi : 10 000 €</li><li>Taux de distribution moyen : 4,5 % par an</li><li>Revenus bruts cumulés sur 8 ans (intérêts composés) : ≈ 4 216 €</li></ul><h3>Performance totale</h3><div class="s2m-note"><ul><li>Plus-value nette (après impôts et forfait 7,5 %) : <strong>≈ 851 €</strong></li><li>Revenus cumulés (bruts, avant IR sur revenus fonciers) : <strong>≈ 4 216 €</strong></li><li><strong>Gain total brut : ≈ 5 067 €</strong></li><li><strong>TRI indicatif (avant fiscalité revenus) : ≈ 5,8 %</strong></li></ul></div><p>Ce calcul est illustratif. Il intègre les intérêts composés pour les revenus mais ne tient pas compte de la fiscalité sur les revenus annuels distribués (TMI + PS de 17,2 %), ni de l'inflation. Le TRI réel dépend de votre tranche marginale d'imposition.</p><h2>Points pratiques et stratégie</h2><h3>Durée de détention et fiscalité</h3><table class="s2m-table"><thead><tr><th>Durée</th><th>Abattement IR</th><th>Abattement PS</th><th>Taux effectif global</th></tr></thead><tbody><tr><td>&lt; 6 ans</td><td>0 %</td><td>0 %</td><td>36,2 % (+ surtaxe éventuelle)</td></tr><tr><td>10 ans</td><td>30 %</td><td>8,25 %</td><td>≈ 29,00 %</td></tr><tr><td>15 ans</td><td>60 %</td><td>16,50 %</td><td>≈ 21,95 %</td></tr><tr><td>20 ans</td><td>90 %</td><td>24,75 %</td><td>≈ 14,82 %</td></tr><tr><td>≥ 22 ans</td><td>100 %</td><td>28 %</td><td>≈ 12,38 % (PS seuls)</td></tr><tr><td>25 ans</td><td>100 %</td><td>55 %</td><td>≈ 7,74 % (PS seuls)</td></tr><tr><td>≥ 30 ans</td><td>100 %</td><td>100 %</td><td><strong>0 %</strong></td></tr></tbody></table><h3>Arbitrages et revente</h3><p>Avant de céder vos parts de SCPI, comparez :</p><ul><li>Le coût fiscal total de la cession (IR + PS + surtaxe éventuelle)</li><li>Les frais de vente (commission plateforme, frais de notaire, frais société de gestion)</li><li>La performance potentielle d'un réinvestissement</li><li>L'intérêt d'attendre la prochaine tranche d'abattement (chaque année compte entre la 6ᵉ et la 22ᵉ année)</li><li>L'éventualité d'une réforme future du régime des plus-values (voir encadré ci-dessous)</li></ul><p>Utilisez le <a href="/simulateurs/arbitrage-scpi">simulateur d'arbitrage SCPI</a> pour évaluer votre situation.</p><div class="s2m-note s2m-warning"><p><strong>Réforme en discussion :</strong> Lors de l'examen du PLF 2026, l'Assemblée nationale avait adopté un amendement ramenant l'exonération IR de 22 à 17 ans, et le Sénat avait voté une refonte radicale (suppression des abattements, coefficient d'érosion monétaire, taux réduits). <strong>Aucune de ces mesures n'a été retenue dans le texte final adopté le 2 février 2026 par 49.3.</strong> Le régime actuel (22 ans IR / 30 ans PS) reste donc inchangé. Toutefois, une réforme reste envisageable dans un futur PLF.</p></div><h2>Vente sur marché secondaire : frais &amp; net vendeur</h2><p>En cas de cession de gré à gré via 2ndMarket, plusieurs postes de SCPI : dispositifs en vigueur en 2026</h2><div class="s2m-note s2m-danger"><p><strong>Fin du Pinel :</strong> Le dispositif Pinel a pris fin le <strong>31 décembre 2024</strong>. Il n'est plus possible de souscrire à de nouvelles SCPI Pinel depuis le 1ᵉʳ janvier 2025. Les souscriptions réalisées avant cette date continuent de produire leurs effets fiscaux jusqu'au terme de l'engagement.</p></div><h3>Dispositif Jeanbrun (PLF 2026) — Nouveau</h3><ul><li><strong>Principe :</strong> Amortissement fiscal pour les bailleurs privés louant un logement nu avec un bail d'au moins 9 ans, sur tout le territoire</li><li><strong>Avantage :</strong> Amortissement de 2 % à 3,5 % par an sur le prix du bien + déficit foncier imputable jusqu'à 21 400 € par an (au lieu de 10 700 €)</li><li><strong>Condition :</strong> Loyers plafonnés et ressources locataires encadrées</li><li><strong>Statut :</strong> Adopté dans le PLF 2026 le 2 février 2026, en attente de décrets d'application et de validation par le Conseil constitutionnel</li></ul><h3>SCPI Malraux : pour TMI élevées</h3><ul><li><strong>Profil :</strong> TMI 41–45 %, ticket d'entrée ≥ 50 000 €</li><li><strong>Réduction d'impôt :</strong> jusqu'à 30 % du montant investi dans certains secteurs sauvegardés, dans la limite des plafonds légaux</li><li><strong>Points d'attention :</strong> 2–3 ans de travaux sans revenus, frais d'entrée 10–12 %, revente moins liquide après la période fiscale</li></ul><h3>SCPI Denormandie : villes moyennes</h3><ul><li><strong>Principe :</strong> Réduction d'impôt pour investissement dans l'ancien avec rénovation, dans des villes éligibles</li><li><strong>Avantages :</strong> Prix d'achat plus accessibles, réduction pouvant atteindre 21 % sur 12 ans</li><li><strong>Contraintes :</strong> Travaux représentant ≥ 25 % du coût total, encadrement des loyers</li></ul><h3>SCPI Déficit foncier : pour investisseurs déjà bailleurs</h3><ul><li><strong>Déficit imputable :</strong> jusqu'à 10 700 € par an sur le revenu global (21 400 € avec Jeanbrun si applicable)</li><li><strong>Report :</strong> déficit excédentaire reportable sur revenus fonciers pendant 10 ans</li><li><strong>Pertinent si :</strong> revenus fonciers déjà significatifs (&gt; En souscription neuve directement auprès de la société de gestion, il n'y a pas de droits d'enregistrement sur les parts.</p></div><h2>Questions fréquentes</h2><h3>Comment se calcule la plus-value sur des parts de SCPI ?</h3><p>Plus-value brute = Prix de cession − (Prix d'acquisition + Frais d'acquisition). Le prix d'acquisition peut être majoré d'un forfait de 7,5 % au titre des frais d'acquisition. La plus-value brute est ensuite soumise à l'IR (19 %) et aux PS (17,2 %) après application des abattements pour durée de détention. Si la plus-value nette dépasse 50 000 €, une surtaxe de 2 % à 6 % s'ajoute.</p><h3>Quand est-on totalement exonéré ?</h3><p>L'impôt sur le revenu est nul après <strong>22 ans</strong> de détention. Les prélèvements sociaux sont totalement supprimés après <strong>30 ans</strong>. Entre ces deux seuils, seuls les PS restent dus (au taux dégressif).</p><h3>Peut-on appliquer le forfait de 7,5 % pour les frais d'acquisition ?</h3><p>Oui. Si vous ne pouvez justifier de frais d'acquisition supérieurs (droits d'enregistrement, frais de notaire à l'achat), vous pouvez opter pour une majoration forfaitaire de 7,5 % du prix d'achat. Ce forfait est souvent avantageux sur le marché secondaire où les droits d'enregistrement de 5 % sont inférieurs au forfait.</p><h3>Les prélèvements sociaux ont-ils augmenté en 2026 pour les SCPI ?</h3><p>La LFSS 2026 a relevé les PS à 18,6 % pour certains revenus du capital (dividendes, plus-values mobilières, LMNP). Cependant, les <strong>plus-values immobilières</strong> (y compris sur parts de SCPI) et les <strong>revenus fonciers</strong> issus de location nue restent à <strong>17,2 %</strong>. En revanche, les revenus de LMNP (meublés non professionnels) sont soumis au nouveau taux de 18,6 %.</p><h3>Les frais de vente réduisent-ils la base imposable ?</h3><p>Les <strong>frais d'acquisition initiaux</strong> augmentent le prix de revient et réduisent la plus-value imposable. Les <strong>frais de cession</strong> (commissions, frais de notaire à la vente) diminuent le net vendeur mais n'entrent <strong>pas</strong> dans le calcul de la plus-value imposable.</p><h3>La décote sur le marché secondaire impacte-t-elle la fiscalité ?</h3><p>Oui. Une vente à décote réduit mécaniquement la plus-value, voire peut créer une moins-value. En cas de moins-value, aucun impôt n'est dû, mais la moins-value immobilière <strong>ne peut pas être imputée</strong> sur d'autres plus-values (contrairement aux moins-values mobilières).</p><h3>Le PLF 2026 a-t-il modifié le régime des plus-values immobilières ?</h3><p>Non. Le texte adopté le 2 février 2026 n'a retenu <strong>aucune modification</strong> du régime des plus-values immobilières. Le barème d'abattement (22 ans IR / 30 ans PS) et les taux (19 % IR + 17,2 % PS) restent inchangés. Les amendements discutés (passage à 17 ans, suppression des abattements avec érosion monétaire) n'ont pas été intégrés dans la version finale.</p><h3>Qu'est-ce que le dispositif Jeanbrun ?</h3><p>Adopté dans le PLF 2026, le dispositif Jeanbrun crée un statut de bailleur privé permettant un amortissement fiscal et un déficit foncier imputable jusqu'à 21 400 €/an, pour les locations nues avec bail d'au moins 9 ans. Il remplace le Pinel comme principal outil d'incitation à l'investissement locatif.
